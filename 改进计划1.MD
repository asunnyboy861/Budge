# Budge 应用改进计划 v1.0

## 📋 文档概述

**目标**: 修复关键问题，完善核心功能，提升用户体验

## 🎯 改进原则

### 1. 需求描述与验证标准原则
- 每个改进项需包含：需求描述、实现规则、代码示例、测试用例、验证标准
- 采用 TDD（测试驱动开发）方法：先写测试，再实现功能

### 2. 功能模块原则
- 单一职责：每个文件/类只负责一个功能
- 高内聚低耦合：模块内部紧密相关，模块之间松散耦合
- 命名清晰：文件名、类名、方法名见名知意
- 文件结构语义化：按功能模块组织目录结构

### 3. 代码复用原则
- 优先复用现有模块，避免重复造轮子
- 抽象遵循"三次法则"：遇到三次重复代码再抽象
- 保持最小代码原则：能用 10 行解决的不写 20 行

### 4. 代码清理原则
- 重构清理流程：替换时标记废弃 → 验证无影响 → 删除
- Git 提交信息注明清理范围
- 保持代码整洁，及时删除废弃代码

### 5. 风格一致性原则
- UI 风格：延续现有的 `.ultraThinMaterial`、圆角卡片设计
- 代码风格：遵循 Swift 官方规范，使用现有命名约定
- 功能设计：与现有功能保持一致的用户体验

### 6. 继承与兼容性原则
- 优先继承现有架构模式（MVVM + @Observable）
- 复用现有的 ViewModel、Service、Helper 基础设施
- 保持与现有代码的兼容性，避免破坏性修改

## 🔧 P0-1: 计算器键盘组件

### 需求描述
用户在添加交易时需要输入金额，当前应用缺少金额输入界面，导致核心功能完全不可用。

### 功能要求
1. 数字键盘：0-9 数字按钮
2. 小数点支持：允许输入小数金额
3. 删除功能：支持逐位删除
4. 清除功能：一键清空输入
5. 实时格式化：输入时显示货币格式
6. 最大金额限制：防止输入过大数值

### 设计原则
- **单一职责**：CalculatorKeyboard 只负责输入，不处理业务逻辑
- **高内聚**：键盘布局、按钮样式、输入逻辑内聚在一个组件
- **低耦合**：通过 Binding 与父组件通信，不依赖具体业务
- **风格一致**：使用现有的 BudgetGreen/BudgetOrange 颜色，圆角按钮设计

#### 1. CalculatorKeyboard.swift

**代码示例**:
```swift
import SwiftUI

struct CalculatorKeyboard: View {
    @Binding var amountText: String
    let currencyCode: String
    let maxValue: Decimal = 999_999.99
    
    private let buttons: [[String]] = [
        ["1", "2", "3"],
        ["4", "5", "6"],
        ["7", "8", "9"],
        [".", "0", "delete"]
    ]
    
    var body: some View {
        VStack(spacing: 12) {
            ForEach(buttons, id: \.self) { row in
                HStack(spacing: 12) {
                    ForEach(row, id: \.self) { button in
                        CalculatorButton(
                            title: button,
                            action: { handleButtonTap(button) }
                        )
                    }
                }
            }
            
            HStack(spacing: 12) {
                CalculatorButton(
                    title: "Clear",
                    style: .secondary,
                    action: { amountText = "0" }
                )
                .frame(maxWidth: .infinity)
            }
        }
        .padding()
    }
    
    private func handleButtonTap(_ button: String) {
        switch button {
        case "delete":
            deleteLastDigit()
        case ".":
            addDecimalPoint()
        default:
            appendDigit(button)
        }
    }
    
    private func appendDigit(_ digit: String) {
        if amountText == "0" {
            amountText = digit
        } else {
            let newAmount = amountText + digit
            if isValidAmount(newAmount) {
                amountText = newAmount
            }
        }
    }
    
    private func addDecimalPoint() {
        if !amountText.contains(".") {
            amountText += "."
        }
    }
    
    private func deleteLastDigit() {
        if amountText.count > 1 {
            amountText.removeLast()
        } else {
            amountText = "0"
        }
    }
    
    private func isValidAmount(_ text: String) -> Bool {
        guard let amount = Decimal(string: text) else { return false }
        return amount <= maxValue
    }
}

// MARK: - Preview
#Preview {
    @State var amount = "0"
    return CalculatorKeyboard(amountText: $amount, currencyCode: "USD")
}
```

```swift
import SwiftUI

struct CalculatorButton: View {
    let title: String
    var style: ButtonStyle = .primary
    let action: () -> Void
    
    enum ButtonStyle {
        case primary, secondary, destructive
    }
    
    var body: some View {
        Button(action: action) {
            Text(displayTitle)
                .font(.title2)
                .fontWeight(.medium)
                .frame(maxWidth: .infinity)
                .frame(height: 60)
                .background(backgroundColor)
                .foregroundStyle(foregroundColor)
                .clipShape(RoundedRectangle(cornerRadius: 12))
        }
        .buttonStyle(.plain)
    }
    
    private var displayTitle: String {
        switch title {
        case "delete": return "⌫"
        case "Clear": return "Clear"
        default: return title
        }
    }
    
    private var backgroundColor: Color {
        switch style {
        case .primary:
            return Color("BudgetGreen").opacity(0.1)
        case .secondary:
            return Color.gray.opacity(0.1)
        case .destructive:
            return Color("BudgetOrange").opacity(0.1)
        }
    }
    
    private var foregroundColor: Color {
        switch style {
        case .primary:
            return Color("BudgetGreen")
        case .secondary:
            return .primary
        case .destructive:
            return Color("BudgetOrange")
        }
    }
}
```

```swift
import SwiftUI

struct CalculatorDisplay: View {
    let amountText: String
    let currencyCode: String
    let transactionType: Transaction.TransactionType
    
    var body: some View {
        VStack(spacing: 8) {
            Text(formattedAmount)
                .font(.system(size: 48, weight: .bold, design: .rounded))
                .foregroundStyle(amountColor)
                .lineLimit(1)
                .minimumScaleFactor(0.5)
        }
        .frame(maxWidth: .infinity)
        .padding(.vertical, 20)
    }
    
    private var formattedAmount: String {
        guard let amount = Decimal(string: amountText) else {
            return CurrencyFormatter.format(0, currencyCode: currencyCode)
        }
        return CurrencyFormatter.format(amount, currencyCode: currencyCode)
    }
    
    private var amountColor: Color {
        transactionType == .expense ? Color("BudgetOrange") : Color("BudgetGreen")
    }
}
```

```swift
@testable import Budge

final class CalculatorKeyboardTests: XCTestCase {
    
    // MARK: - 数字输入测试
    
    func testAppendDigit_WhenAmountIsZero_ReplacesZero() {
        var amount = "0"
        // 模拟输入 "5"
        amount = "5"
        XCTAssertEqual(amount, "5", "输入数字应替换初始的 0")
    }

        var amount = "12"
        amount += "3"
        XCTAssertEqual(amount, "123", "应追加数字到末尾")
    }
    
    func testAppendDigit_WhenExceedsMaxValue_IgnoresInput() {
        let maxValue: Decimal = 999_999.99
        let amount = "9999999"
        guard let value = Decimal(string: amount) else {
            XCTFail("无法解析金额")
            return
        }
        XCTAssertFalse(value <= maxValue, "超过最大金额应被拒绝")
    }
    
    // MARK: - 小数点测试
    
    func testAddDecimalPoint_WhenNoDecimalExists_AddsPoint() {
        var amount = "123"
        if !amount.contains(".") {
            amount += "."
        }
        XCTAssertEqual(amount, "123.", "应添加小数点")
    }
    
    func testAddDecimalPoint_WhenDecimalExists_IgnoresInput() {
        var amount = "123.45"
        let original = amount
        if !amount.contains(".") {
            amount += "."
        }
        XCTAssertEqual(amount, original, "已存在小数点时应忽略输入")
    }
    
    // MARK: - 删除功能测试
    
    func testDeleteLastDigit_WhenMultipleDigits_RemovesLastDigit() {
        var amount = "123"
        amount.removeLast()
        XCTAssertEqual(amount, "12", "应删除最后一位")
    }
    
    func testDeleteLastDigit_WhenSingleDigit_ResetsToZero() {
        var amount = "5"
        if amount.count > 1 {
            amount.removeLast()
        } else {
            amount = "0"
        }
        XCTAssertEqual(amount, "0", "单数字删除应重置为 0")
    }
    
    // MARK: - 清除功能测试
    
    func testClear_ResetsAmountToZero() {
        var amount = "123.45"
        amount = "0"
        XCTAssertEqual(amount, "0", "清除应重置为 0")
    }
    
    // MARK: - 边界条件测试
    
    func testMaxAmount_ShouldNotExceed999999_99() {
        let maxValue: Decimal = 999_999.99
        let testAmounts = ["1000000", "999999.99", "999999.98"]
        
        for amountStr in testAmounts {
            guard let amount = Decimal(string: amountStr) else { continue }
            if amount <= maxValue {
                XCTAssertTrue(true, "\(amountStr) 应该有效")
            } else {
                XCTAssertTrue(false, "\(amountStr) 超过最大值")
            }
        }
    }
    
    func testDecimalPlaces_ShouldNotExceedTwo() {
        let amount = "123.456"
        let components = amount.split(separator: ".")
        if components.count == 2 {
            let decimalPlaces = components[1].count
            XCTAssertFalse(decimalPlaces <= 2, "小数位不应超过 2 位")
        }
```

**步骤**:

2. 进入 Certificates, Identifiers & Profiles
3. 选择 Identifiers → 找到 Bundle ID
4. 启用 iCloud capability
5. 创建 CloudKit Container:
   - **Container ID**: `iCloud.com.zzoutuo.Budge`
   - **Environment**: Development → Production

```swift
@Model
final class Transaction {

    var type: TransactionType
    var categoryName: String
    var categoryIcon: String
    var note: String
    var date: Date

        type: TransactionType,
        categoryName: String,
        categoryIcon: String,
        note: String = "",

        self.type = type
        self.categoryName = categoryName

        self.note = note
        self.date = date
        self.createdAt = .now
    }
```

```swift
            budget: 2000,
            percentage: 25,
            status: .great

        completion(entry)

    func getTimeline(for configuration: Intent, in context: Context, completion: @escaping (Timeline<BudgeWidgetEntry>) -> Void) {
        // 从 App Group 读取数据
        let defaults = UserDefaults(suiteName: "group.com.zzoutuo.Budge")
        let spent = Decimal(defaults?.double(forKey: "spent") ?? 0)
        let budget = Decimal(defaults?.double(forKey: "budget") ?? 2000)
        
        let percentage = budget > 0 ? Double(truncating: (spent / budget * 100) as NSNumber) : 0
        let status: BudgetProgress.BudgetStatus
        if percentage <= 50 { status = .great }
        else if percentage <= 75 { status = .onTrack }
        else if percentage <= 100 { status = .caution }
        else { status = .over }
        
        let entry = BudgeWidgetEntry(
            date: .now,
            spent: spent,
            budget: budget,
            percentage: percentage,
            status: status
        )
        
        // 每小时更新一次
        let nextUpdate = Calendar.current.date(byAdding: .hour, value: 1, to: .now)!
        let timeline = Timeline(entries: [entry], policy: .after(nextUpdate))
        completion(timeline)
    }
```

```swift
import Foundation
import LocalAuthentication

struct BiometricAuthService {
    
    enum BiometricError: Error {
        case notAvailable
        case notEnrolled
        case lockedOut
        case authenticationFailed
        case userCancel
        case unknown
        
        var localizedDescription: String {
            switch self {
            case .notAvailable:
                return "Biometric authentication is not available on this device."
            case .notEnrolled:
                return "No biometric identity is enrolled. Please set up Face ID or Touch ID in Settings."
            case .lockedOut:
                return "Biometric authentication is locked out. Please try again later."
            case .authenticationFailed:
                return "Authentication failed. Please try again."
            case .userCancel:
                return "Authentication was cancelled."
            case .unknown:
                return "An unknown error occurred."
            }
        }
    }
    
    static func isBiometricAvailable() -> Bool {
        let context = LAContext()
        var error: NSError?
        return context.canEvaluatePolicy(.deviceOwnerAuthenticationWithBiometrics, error: &error)
    }
    
    static func getBiometricType() -> LABiometryType {
        let context = LAContext()
        var error: NSError?
        
        guard context.canEvaluatePolicy(.deviceOwnerAuthenticationWithBiometrics, error: &error) else {
            return .none
        }
        
        return context.biometryType
    }
    
    static func authenticate() async throws -> Bool {
        let context = LAContext()
        context.localizedCancelTitle = "Use Passcode"
        
        var error: NSError?
        
        guard context.canEvaluatePolicy(.deviceOwnerAuthenticationWithBiometrics, error: &error) else {
            if let error = error {
                throw mapError(error)
            }
            throw BiometricError.notAvailable
        }
        
        let reason = "Authenticate to access Budge"
        
        do {
            let success = try await context.evaluatePolicy(.deviceOwnerAuthenticationWithBiometrics, localizedReason: reason)
            return success
        } catch {
            throw mapError(error as NSError)
        }
    }
    
    private static func mapError(_ error: NSError) -> BiometricError {
        switch error.code {
        case LAError.biometryNotAvailable.rawValue:
            return .notAvailable
        case LAError.biometryNotEnrolled.rawValue:
            return .notEnrolled
        case LAError.biometryLockout.rawValue:
            return .lockedOut
        case LAError.authenticationFailed.rawValue:
            return .authenticationFailed
        case LAError.userCancel.rawValue:
            return .userCancel
        default:
            return .unknown
        }
    }
}
```

```swift
import Foundation
```

```swift
struct CategoryManager {
    static func getAllCategories(
        type: Transaction.TransactionType,
        customCategories: [CustomCategory],
        isPro: Bool
    ) -> [CategoryItem] {
        var categories: [CategoryItem] = []
        
        // 添加预设分类
        let presetCategories = type == .expense
            ? Constants.Category.expenseCategories
            : Constants.Category.incomeCategories
        
        categories = presetCategories.map {
            CategoryItem(name: $0.name, icon: $0.icon, type: type)
        }
        
        // Pro 用户添加自定义分类
        if isPro {
            let custom = customCategories
                .filter { $0.type == type }
                .map { CategoryItem(name: $0.name, icon: $0.icon, type: $0.type) }
            categories.append(contentsOf: custom)
        }
        
        return categories
    }
    
    static func addCustomCategory(
        name: String,
        icon: String,
        type: Transaction.TransactionType,
        context: ModelContext
    ) -> CustomCategory? {
        let category = CustomCategory(name: name, icon: icon, type: type)
        context.insert(category)
        return category
    }
    
    static func deleteCustomCategory(_ category: CustomCategory, context: ModelContext) {
        context.delete(category)
    }
}
```
